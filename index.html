// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

contract DecentralizedFuture {
    address public owner;
    uint256 public projectCount;
    
    struct Project {
        uint256 id;
        string name;
        string description;
        uint256 fundingGoal;
        uint256 currentFunding;
    }
    
    mapping(uint256 => Project) public projects;
    
    event ProjectCreated(uint256 id, string name);
    event ProjectFunded(uint256 id, uint256 amount);
    
    constructor() {
        owner = msg.sender;
    }
    
    function createProject(
        string memory _name,
        string memory _description,
        uint256 _fundingGoal
    ) external {
        projectCount++;
        projects[projectCount] = Project({
            id: projectCount,
            name: _name,
            description: _description,
            fundingGoal: _fundingGoal,
            currentFunding: 0
        });
        
        emit ProjectCreated(projectCount, _name);
    }
    
    function fundProject(uint256 _id) external payable {
        require(_id > 0 && _id <= projectCount, "Invalid project ID");
        require(msg.value > 0, "Must send ETH to fund");
        
        Project storage project = projects[_id];
        project.currentFunding += msg.value;
        
        emit ProjectFunded(_id, msg.value);
    }
}
